项目完整源码

基于YOLO目标检测的工人出入产线警报系统 · Python 实现

以下为该项目的完整 Python 源代码。代码包含模型加载、自定义多边形区域定义、IoU 交集判定函数以及逐帧推理与结果保存逻辑。

导入依赖 from ultralytics import YOLO, cv2, numpy, os
# 导入所需库 from ultralytics import YOLO import cv2 import numpy as np import os
模型加载 model = YOLO("best2.pt")
# 加载训练好的 YOLO 模型权重 model = YOLO("best2.pt")
自定义多边形区域（产线边界） area = np.array([...])
# 在摄像头画面中手动标定的产线多边形顶点坐标 area = np.array([ (635, 90), (547, 326), (519, 327), (499, 388), (407, 622), (361, 752), (323, 1050), (1417, 1050), (1347, 830), (1385, 824), (1277, 598), (1166, 352), (1124, 258), (1086, 259), (1007, 90) ], dtype=np.int32) # 人物类别 ID（COCO 数据集中 person = 0） people_class_id = 0
IoU 交集判定函数 is_overlapping(rect, polygon)
# 判断矩形框与多边形是否有交集（IoU > 0） def is_overlapping(rect, polygon): x1, y1, x2, y2 = rect rect_pts = np.array([ [x1, y1], [x2, y1], [x2, y2], [x1, y2] ], dtype=np.float32) poly_pts = polygon.astype(np.float32) ret, inter_pts = cv2.intersectConvexConvex(rect_pts, poly_pts) if ret > 0: return True if inter_pts is not None and len(inter_pts) > 0: return True return False
推理与判定主循环 results = model.predict(...)
# 对指定目录中的图像进行流式推理 results = model.predict( source=r"C:\Users\19511\Desktop\Temp", stream=True, save=False, show=False, conf=0.3, iou=0 ) output_dir = "runs/detect/predict" os.makedirs(output_dir, exist_ok=True) for idx, result in enumerate(results): boxes = result.boxes # 若无检测框，则仅绘制多边形并保存原图 if boxes is None or len(boxes) == 0: img = result.orig_img.copy() cv2.polylines(img, [area], True, (255, 0, 0), 2) save_path = os.path.join(output_dir, os.path.basename(result.path)) cv2.imwrite(save_path, img) continue # 提取所有人物检测框 people_boxes = [] for box in boxes: cls = int(box.cls[0]) if cls != people_class_id: continue x1, y1, x2, y2 = box.xyxy[0].tolist() conf = float(box.conf[0]) people_boxes.append({'box': [x1, y1, x2, y2], 'confidence': conf}) # 判定每个工人是否与多边形区域重叠 overlap_found = False persons_inside = [] for person in people_boxes: rect = person['box'] if is_overlapping(rect, area): overlap_found = True persons_inside.append(rect) # 绘制结果图像 img_with_boxes = result.plot() cv2.polylines(img_with_boxes, [area], True, (255, 0, 0), 2) if overlap_found: cv2.putText(img_with_boxes, "PERSON ENTERED AREA!", (50, 50), cv2.FONT_HERSHEY_SIMPLEX, 1.5, (0, 0, 255), 4) for rect in persons_inside: x1, y1, x2, y2 = map(int, rect) cv2.rectangle(img_with_boxes, (x1, y1), (x2, y2), (0, 0, 255), 3) cv2.putText(img_with_boxes, "INSIDE", (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)
结果保存 cv2.imwrite(...)
# 根据是否触发警报，添加前缀保存 base_name = os.path.basename(result.path) if overlap_found: save_name = f"inside_{base_name}" else: save_name = base_name save_path = os.path.join(output_dir, save_name) cv2.imwrite(save_path, img_with_boxes) # 全部处理完毕后输出提示 print("\nDone")
补充说明
  • 模型权重： best2.pt 为经过三个版本迭代训练后的 YOLOv11n 模型，对工人和产线的识别精度满足部署要求。
  • 多边形区域：坐标值针对特定摄像头视角标定，若更换摄像头位置需重新标定。
  • 置信度阈值： conf=0.3 为平衡检测率与误报率的经验值，可根据实际场景调整。
  • 输出目录：所有结果图像保存在 runs/detect/predict/ 下，触发警报的文件名以 inside_ 为前缀。
