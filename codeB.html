基于YOLO与PaddleOCR的板坯号识别系统 · Python 实现

以下为该项目的核心 Python 代码，主要实现 YOLO 板坯区域检测、OpenCV 视频流处理以及交互式控制（暂停、快进、慢放等）。完整代码还包含 PaddleOCR 文字识别部分，此处仅展示视频处理与目标检测逻辑。

导入依赖 import cv2, from ultralytics import YOLO
# 导入所需库 import cv2 from ultralytics import YOLO

# 加载训练好的 YOLO 板坯检测模型 model = YOLO("Blockbest.pt")
视频路径与初始化 cv2.VideoCapture(...)
video_path = r"C:\Users\19511\Desktop\板坯识别\C6\C6看板号150.58_C6看板号150.58_20260715050000_20260715053000.avi" cap = cv2.VideoCapture(video_path) if not cap.isOpened(): print("无法打开视频！") exit() fps = int(cap.get(cv2.CAP_PROP_FPS)) total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT)) width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)) height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
交互控制参数 pause, skip_frames, current_frame
pause = False # 暂停标志 skip_frames = 1 # 每次读取的帧数（1为正常，>1为快进） current_frame = 0 # 当前播放到的帧索引（用于显示进度） print(" 空格 - 暂停/继续") print(" →/F - 快进（每次跳2帧）") print(" ←/S - 慢放（恢复1帧）") print(" R - 重置速度") print(" Q/ESC - 退出")
推理主循环 while True: ...
while True: if not pause: # 实现快进：跳过 skip_frames-1 帧 for _ in range(skip_frames - 1): ret = cap.grab() if not ret: break ret, frame = cap.read() if not ret: print("视频播放完毕。") break current_frame += skip_frames # YOLO 推理并绘制结果 results = model(frame, conf=0.5) annotated_frame = results[0].plot() # 叠加帧信息与速度 cv2.putText(annotated_frame, f"Frame: {current_frame}/{total_frames}", (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2) cv2.putText(annotated_frame, f"Speed: {skip_frames}x", (10, 60), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2) cv2.imshow("YOLO Detection", annotated_frame)
键盘事件 & 资源释放 cv2.waitKey() / cap.release()
key = cv2.waitKey(1) & 0xFF if key == ord('q') or key == 27: break elif key == 32: pause = not pause print("暂停" if pause else "继续") elif key == ord('f') or key == 255: skip_frames = min(skip_frames * 2, 16) print(f"快进 x{skip_frames}") elif key == ord('s') or key == 81: skip_frames = max(skip_frames // 2, 1) print(f"速度 x{skip_frames}") elif key == ord('r'): skip_frames = 1 print("速度重置为正常") cap.release() cv2.destroyAllWindows()
补充说明
  • 模型权重： Blockbest.pt 为基于 YOLOv11n 在板坯数据集上微调后的模型，对板坯正面区域具有较高的定位精度。
  • 视频路径： 示例中使用的视频文件为产线现场采集的 .avi 格式文件，需根据实际路径修改。
  • 置信度阈值： conf=0.5 为平衡检测率与误检率的经验值，可根据实际场景调整。
  • 交互控制： 支持空格暂停/继续、F/→ 快进、S/← 慢放、R 重置速度，便于人工复核检测结果。
  • 后续处理： 在实际部署中，检测到的板坯区域会裁剪后送入 PaddleOCR 进行文字识别，完整流程请参考项目说明。
